cmake_minimum_required(VERSION 3.18)
project(terminal_notifier VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)

list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/usr")
find_package(Threads REQUIRED)
find_path(ICAL_INCLUDE_DIR libical/ical.h REQUIRED)
find_library(ICAL_LIBRARY NAMES libical.so.3 ical REQUIRED)
find_path(CURL_INCLUDE_DIR curl/curl.h PATH_SUFFIXES x86_64-linux-gnu REQUIRED)
find_library(CURL_LIBRARY NAMES libcurl.so.4 curl REQUIRED)
find_path(DBUS_INCLUDE_DIR dbus/dbus.h PATH_SUFFIXES dbus-1.0 REQUIRED)
find_path(DBUS_ARCH_INCLUDE_DIR dbus/dbus-arch-deps.h
    PATHS /usr/lib /usr/lib/x86_64-linux-gnu
          "${CMAKE_SOURCE_DIR}/.deps/usr/lib/x86_64-linux-gnu"
    PATH_SUFFIXES dbus-1.0/include REQUIRED)
find_library(DBUS_LIBRARY NAMES libdbus-1.so.3 dbus-1 REQUIRED)
find_library(NCURSES_LIBRARY NAMES ncursesw REQUIRED)
find_path(NCURSES_INCLUDE_DIR ncursesw/curses.h REQUIRED)

add_library(notifier_core src/core.cpp src/calendar.cpp)
target_include_directories(notifier_core PUBLIC src PRIVATE "${ICAL_INCLUDE_DIR}" "${CURL_INCLUDE_DIR}")
target_link_libraries(notifier_core PUBLIC "${ICAL_LIBRARY}" "${CURL_LIBRARY}")

add_library(notifier_dbus src/notifications.cpp)
target_include_directories(notifier_dbus PUBLIC src PRIVATE "${DBUS_INCLUDE_DIR}" "${DBUS_ARCH_INCLUDE_DIR}")
target_link_libraries(notifier_dbus PUBLIC notifier_core "${DBUS_LIBRARY}")
add_executable(terminal-notifier src/main.cpp)
target_include_directories(terminal-notifier PRIVATE "${NCURSES_INCLUDE_DIR}" "${CURL_INCLUDE_DIR}")
target_link_libraries(terminal-notifier PRIVATE notifier_dbus "${NCURSES_LIBRARY}" Threads::Threads)
install(TARGETS terminal-notifier RUNTIME DESTINATION bin)

enable_testing()
add_executable(core-test tests/core_test.cpp)
target_link_libraries(core-test PRIVATE notifier_core)
add_test(NAME notification-routing COMMAND core-test)
add_executable(calendar-test tests/calendar_test.cpp)
target_link_libraries(calendar-test PRIVATE notifier_core)
add_test(NAME calendar-parsing COMMAND calendar-test)
find_program(DBUS_RUN_SESSION dbus-run-session)
if(DBUS_RUN_SESSION)
    add_executable(notifications-test tests/notifications_test.cpp)
    target_include_directories(notifications-test PRIVATE "${DBUS_INCLUDE_DIR}" "${DBUS_ARCH_INCLUDE_DIR}")
    target_link_libraries(notifications-test PRIVATE notifier_dbus)
    add_test(NAME dbus-monitoring COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:notifications-test>)
    set_tests_properties(dbus-monitoring PROPERTIES TIMEOUT 10)
endif()
add_executable(terminal-test tests/terminal_test.cpp)
find_library(UTIL_LIBRARY NAMES util)
if(UTIL_LIBRARY)
    target_link_libraries(terminal-test PRIVATE "${UTIL_LIBRARY}")
endif()
add_test(NAME terminal-panes COMMAND terminal-test $<TARGET_FILE:terminal-notifier>)
set_tests_properties(terminal-panes PROPERTIES TIMEOUT 15)
